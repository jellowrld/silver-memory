    function httpJSON(url, headers = {}) {
        return new Promise((resolve, reject) => {
            GM_xmlhttpRequest({
                method: "GET",
                url,
                headers,
                onload: res => {
                    try { resolve(JSON.parse(res.responseText)); }
                    catch (e) { reject(e); }
                },
                onerror: err => reject(err)
            });
        });
    }

    // =========================
    // GG.DEALS FULL API LOGIC
    // =========================

    async function ggSearchGame(query) {
        const url = `https://api.gg.deals/v1/games/search/?q=${encodeURIComponent(query)}`;
        try {
            const json = await httpJSON(url, GG_HEADERS);
            if (!json || !json.data || !json.data.length) return null;

            // Best match = first item
            return json.data[0];
        } catch (e) {
            console.log("GG Search error:", e);
            return null;
        }
    }

    async function ggGetOffers(gameID) {
        const url = `https://api.gg.deals/v1/games/${gameID}/offers/`;
        try {
            const json = await httpJSON(url, GG_HEADERS);
            if (!json || !json.data || !json.data.length) return [];

            return json.data.map(o => {
                const s = GG_STORE_MAP[o.store_id] || {
                    name: `Store ${o.store_id}`,
                    sellerType: "reseller",
                    redeem: "other",
                    productType: "key"
                };

                return {
                    source: "ggdeals",
                    title: o.title,
                    store: s.name,
                    sellerType: s.sellerType,
                    redeem: s.redeem,
                    productType: s.productType,
                    price: o.price?.amount || null,
                    currency: o.price?.currency || "USD",
                    url: o.url
                };
            });
        } catch (e) {
            console.log("GG Offers error:", e);
            return [];
        }
    }

    async function fetchGGDeals(query) {
        const game = await ggSearchGame(query);
        if (!game) return [];

        const gameID = game.id;
        return await ggGetOffers(gameID);
    }

    // =========================
    // CHEAPSHARK (unchanged)
    // =========================
    async function fetchCheapShark(query) {
        const url = `https://www.cheapshark.com/api/1.0/deals?title=${encodeURIComponent(query)}`;
        try {
            const data = await httpJSON(url);
            return data.map(d => ({
                source: "cheapshark",
                title: d.title,
                store: d.storeID,
                sellerType: "reseller",
                redeem: "multi",
                productType: "key",
                price: parseFloat(d.salePrice),
                currency: "USD",
                url: `https://www.cheapshark.com/redirect?dealID=${d.dealID}`
            }));
        } catch {
            return [];
        }
    }
